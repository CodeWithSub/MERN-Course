// db.collection.updateOne() Update at most a single document that match a specific filter even though multiple documents may match the specific filter

db.students.updateOne({name:"Subhransu"},{$set: {city:"Dhenkanal"}});

// db.collection.updateOne() Update all documents that match a specific filter

db.students.updateMany({marks: {$lt: 75}}, {$set: {city:"Bhubaneswar"}});

// db.collection.replaceOne() Replace at most a single document that matches a single specified filter even though multiple documents may match the specific filter

db.students.replaceOne({name:"Subhransu"},{name:"Adam",age:24,voter:false});
